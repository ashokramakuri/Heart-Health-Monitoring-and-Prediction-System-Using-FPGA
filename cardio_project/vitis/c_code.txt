static void clear_live_window(void)
{
    int i;
    live_pred_count = 0;
    for (i = 0; i < LIVE_VOTE_WINDOW; i++)
        live_pred_window[i] = 0;
    last_live_cls = -1;
}

static int16_t median3(int16_t a, int16_t b, int16_t c)
{
    int16_t t;
    if (a > b) { t = a; a = b; b = t; }
    if (b > c) { t = b; b = c; c = t; }
    if (a > b) { t = a; a = b; b = t; }
    return b;
}

static void free_conn_state(conn_state_t *state)
{
    if (state != NULL)
    {
        if (state->is_dynamic && state->dyn_data != NULL)
            mem_free(state->dyn_data);
        mem_free(state);
    }
}

static void web_push_sample(int sample)
{
    g_web_ring[g_web_ring_write] = sample;
    g_web_ring_write = (g_web_ring_write + 1U) % WEB_RING_SIZE;
    if (g_web_ring_count < WEB_RING_SIZE)
        g_web_ring_count++;
}

static int final_majority_class(void)
{
    int count[NUM_CLASSES] = {0, 0, 0};
    int i;
    int max_class = 0;
    int max_count = 0;

    for (i = 0; i < pred_count; i++)
    {
        if (pred_history[i] >= 0 && pred_history[i] < NUM_CLASSES)
            count[pred_history[i]]++;
    }

    for (i = 0; i < NUM_CLASSES; i++)
    {
        if (count[i] > max_count)
        {
            max_count = count[i];
            max_class = i;
        }
    }

    return max_class;
}

/* =========================
   WEB SERVER CALLBACKS
   ========================= */
static err_t sent_callback(void *arg, struct tcp_pcb *tpcb, u16_t len)
{
    conn_state_t *state = (conn_state_t *)arg;
    LWIP_UNUSED_ARG(len);

    free_conn_state(state);
    tcp_arg(tpcb, NULL);
    tcp_sent(tpcb, NULL);
    tcp_close(tpcb);
    return ERR_OK;
}

static err_t recv_callback(void *arg, struct tcp_pcb *tpcb,
                           struct pbuf *p, err_t err)
{
    char *req;
    conn_state_t *state;
    char buffer[WEB_RESP_SIZE];
    char sample_str[700];
    int n;
    err_t wr_err;

    LWIP_UNUSED_ARG(arg);
    LWIP_UNUSED_ARG(err);

    if (p == NULL)
    {
        tcp_close(tpcb);
        return ERR_OK;
    }

    req = (char *)p->payload;

    state = (conn_state_t *)mem_malloc(sizeof(conn_state_t));
    if (state == NULL)
    {
        pbuf_free(p);
        tcp_abort(tpcb);
        return ERR_ABRT;
    }

    state->dyn_data = NULL;
    state->is_dynamic = 0;

    tcp_arg(tpcb, state);
    tcp_sent(tpcb, sent_callback);

    if ((req != NULL) && (strstr(req, "GET /data") != NULL))
    {
        int i;
        int pos = 0;
        unsigned int count = g_web_ring_count;
        unsigned int write = g_web_ring_write;
        unsigned int send_count = (count < WEB_CHUNK_SAMPLES) ? count : WEB_CHUNK_SAMPLES;
        unsigned int start;

        sample_str[0] = '\0';

        if (send_count > 0U)
        {
            if (write >= send_count)
                start = write - send_count;
            else
                start = WEB_RING_SIZE + write - send_count;

            for (i = 0; i < (int)send_count; i++)
            {
                unsigned int idx = (start + (unsigned int)i) % WEB_RING_SIZE;
                if (i == 0)
                    pos += snprintf(sample_str + pos, sizeof(sample_str) - (size_t)pos, "%d", g_web_ring[idx]);
                else
                    pos += snprintf(sample_str + pos, sizeof(sample_str) - (size_t)pos, "|%d", g_web_ring[idx]);

                if (pos >= (int)sizeof(sample_str) - 16)
                    break;
            }
        }

        n = snprintf(buffer, sizeof(buffer),
                     "HTTP/1.1 200 OK\r\n"
                     "Content-Type: text/plain\r\n"
                     "Connection: close\r\n"
                     "\r\n"
                     "ECG:%d,BPM:%d,CLS:%d,PROC:%d,ELAPSED:%d,SAMPLES:%s",
                     g_ecg_value, g_bpm_value, g_cls_value,
                     g_processing, g_elapsed_sec, sample_str);

        state->dyn_data = (char *)mem_malloc((mem_size_t)(n + 1));
        if (state->dyn_data == NULL)
        {
            free_conn_state(state);
            tcp_arg(tpcb, NULL);
            pbuf_free(p);
            tcp_abort(tpcb);
            return ERR_ABRT;
        }

        memcpy(state->dyn_data, buffer, (size_t)(n + 1));
        state->is_dynamic = 1;

        wr_err = tcp_write(tpcb, state->dyn_data, (u16_t)n, TCP_WRITE_FLAG_COPY);
    }
    else
    {
        wr_err = tcp_write(tpcb, html_page, (u16_t)strlen(html_page), TCP_WRITE_FLAG_COPY);
    }

    if (wr_err != ERR_OK)
    {
        free_conn_state(state);
        tcp_arg(tpcb, NULL);
        pbuf_free(p);
        tcp_abort(tpcb);
        return ERR_ABRT;
    }

    tcp_recved(tpcb, p->tot_len);
    pbuf_free(p);
    tcp_output(tpcb);

    return ERR_OK;
}

static err_t accept_callback(void *arg, struct tcp_pcb *newpcb, err_t err)
{
    LWIP_UNUSED_ARG(arg);
    LWIP_UNUSED_ARG(err);
    tcp_recv(newpcb, recv_callback);
    return ERR_OK;
}

static void start_webserver(void)
{
    struct tcp_pcb *pcb;

    pcb = tcp_new();
    if (pcb == NULL)
    {
        xil_printf("tcp_new failed\r\n");
        return;
    }

    if (tcp_bind(pcb, IP_ADDR_ANY, 80) != ERR_OK)
    {
        xil_printf("tcp_bind failed\r\n");
        tcp_close(pcb);
        return;
    }

    pcb = tcp_listen(pcb);
    if (pcb == NULL)
    {
        xil_printf("tcp_listen failed\r\n");
        return;
    }

    tcp_accept(pcb, accept_callback);
    xil_printf("Webserver started on port 80\r\n");
}

static int Net_Init(void)
{
    ip_addr_t ipaddr, netmask, gw;

    IP4_ADDR(&ipaddr, 192, 168, 1, 10);
    IP4_ADDR(&netmask, 255, 255, 255, 0);
    IP4_ADDR(&gw, 192, 168, 1, 1);

    lwip_init();

    if (!xemac_add(&server_netif, &ipaddr, &netmask, &gw,
                   NULL, XPAR_XEMACPS_0_BASEADDR))
    {
        xil_printf("Error adding network interface\r\n");
        return -1;
    }

    netif_set_default(&server_netif);
    netif_set_up(&server_netif);

    xil_printf("Board IP: 192.168.1.10\r\n");
    return 0;
}

static int I2C_Init(void)
{
    XIicPs_Config *cfg;

    cfg = XIicPs_LookupConfig(IIC_DEVICE_ID);
    if (cfg == NULL)
        return -1;

    if (XIicPs_CfgInitialize(&Iic, cfg, cfg->BaseAddress) != XST_SUCCESS)

    return 0;
}

/* =========================
   PS GPIO
   ========================= */
static int PS_GPIO_Init(void)
{
    XGpioPs_Config *cfg;

    cfg = XGpioPs_LookupConfig(GPIO_DEVICE_ID);
    if (cfg == NULL)
        return -1;

    if (XGpioPs_CfgInitialize(&Gpio, cfg, cfg->BaseAddr) != XST_SUCCESS)
        return -1;

    XGpioPs_SetDirectionPin(&Gpio, LO_PLUS_MIO, 0);
    XGpioPs_SetOutputEnablePin(&Gpio, LO_PLUS_MIO, 0);

    XGpioPs_SetDirectionPin(&Gpio, LO_MINUS_MIO, 0);
    XGpioPs_SetOutputEnablePin(&Gpio, LO_MINUS_MIO, 0);

    return 0;
}

/* =========================
   ADS1115
   ========================= */
static int ADS1115_ReadSingleEnded_AIN0(int16_t *value)
{
    uint8_t config[3] = {0x01, 0xC3, 0x83};
    uint8_t reg;
    uint8_t data[2];
    uint16_t status;

    if (XIicPs_MasterSendPolled(&Iic, config, 3, ADS1115_ADDR) != XST_SUCCESS)
        return -1;
    while (XIicPs_BusIsBusy(&Iic));

    reg = 0x01;
    while (1)
    {
        if (XIicPs_MasterSendPolled(&Iic, &reg, 1, ADS1115_ADDR) != XST_SUCCESS)
            return -1;
        while (XIicPs_BusIsBusy(&Iic));

        if (XIicPs_MasterRecvPolled(&Iic, data, 2, ADS1115_ADDR) != XST_SUCCESS)
            return -1;
        while (XIicPs_BusIsBusy(&Iic));

        status = ((uint16_t)data[0] << 8) | data[1];
        if (status & 0x8000)
            break;
    }

    reg = 0x00;
    if (XIicPs_MasterSendPolled(&Iic, &reg, 1, ADS1115_ADDR) != XST_SUCCESS)
        return -1;
    while (XIicPs_BusIsBusy(&Iic));

    if (XIicPs_MasterRecvPolled(&Iic, data, 2, ADS1115_ADDR) != XST_SUCCESS)
        return -1;
    while (XIicPs_BusIsBusy(&Iic));

    *value = (int16_t)(((uint16_t)data[0] << 8) | data[1]);
    return 0;
}

static int16_t moving_average_filter(int16_t sample)
{
    int i;
    int32_t sum = 0;

    filter_buf[filter_idx] = sample;
    filter_idx = (filter_idx + 1) % FILTER_SIZE;

    if (filter_count < FILTER_SIZE)
        filter_count++;

    for (i = 0; i < filter_count; i++)
        sum += filter_buf[i];

    return (int16_t)(sum / filter_count);
}

static int16_t clean_raw_sample(int16_t raw)
{
    int16_t med;
    int diff;

    if (med_prev1 == 0 && med_prev2 == 0)
    {
        med_prev1 = raw;
        med_prev2 = raw;
    }

    med = median3(med_prev2, med_prev1, raw);
    med_prev2 = med_prev1;
    med_prev1 = raw;

    if (first_sample)
    {
        last_raw_good = med;
        first_sample = 0;
        return med;
    }

    diff = med - last_raw_good;
    if (diff < 0)
        diff = -diff;

    if (diff > MAX_STEP_CHANGE)
        med = last_raw_good;

    if (med >= SATURATION_CLIP_LEVEL)
        med = last_raw_good;

    last_raw_good = med;
    return med;
}
